import 'dart:io';
import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_screenutil/flutter_screenutil.dart';
import 'package:evolve/core/theme/app_theme.dart';
import 'package:evolve/features/auth/presentation/welcome_screen.dart';
import 'package:evolve/features/auth/presentation/widgets/auth_shell.dart';
import 'package:evolve/features/courses/presentation/widgets/lesson_card_surface.dart';
import 'package:evolve/features/onboarding/presentation/steps/name_step.dart';
import 'package:evolve/features/onboarding/presentation/widgets/onboarding_scaffold.dart';

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  final args = await const MethodChannel('holyher/typography')
          .invokeListMethod<String>('previewArguments') ??
      [];
  final route = args
          .where((v) => v.startsWith('--holyher-preview='))
          .firstOrNull
          ?.split('=')
          .last ??
      'welcome';
  runApp(ProviderScope(
      child: ScreenUtilInit(
          designSize: const Size(393, 852),
          minTextAdapt: true,
          builder: (context, child) => MaterialApp(
              debugShowCheckedModeBanner: false,
              theme: AppTheme.darkTheme,
              builder: (context, child) => MediaQuery(
                  data:
                      MediaQuery.of(context).copyWith(disableAnimations: true),
                  child: child!),
              home: route == 'welcome'
                  ? const WelcomeScreen()
                  : route == 'name'
                      ? OnboardingScaffold(
                          pages: [
                              NameStep(
                                  firstName: '',
                                  lastName: '',
                                  accent: AuthTokens.accent,
                                  onChanged: (_, __) {})
                            ],
                          themes: const [
                              AuthTokens.meshTheme
                            ],
                          currentIndex: 0,
                          canContinue: false,
                          continueLabel: 'Continue',
                          onContinue: () {})
                      : Scaffold(
                          body: AuthBackground(
                              child: SafeArea(
                                  child: Padding(
                                      padding: const EdgeInsets.all(24),
                                      child: Column(children: [
                                        const SizedBox(height: 24),
                                        const Text('Lesson preview',
                                            style: TextStyle(
                                                fontSize: 22,
                                                fontWeight: FontWeight.w600)),
                                        const SizedBox(height: 28),
                                        Expanded(
                                            child: LessonCardSurface(
                                                active: true,
                                                child: const Padding(
                                                    padding: EdgeInsets.all(34),
                                                    child: Column(
                                                        crossAxisAlignment:
                                                            CrossAxisAlignment
                                                                .start,
                                                        children: [
                                                          Text('LESSON 01',
                                                              style: TextStyle(
                                                                  fontSize:
                                                                      12)),
                                                          SizedBox(height: 28),
                                                          Text(
                                                              'Learn. Then put it into practice.',
                                                              style: TextStyle(
                                                                  fontSize: 28,
                                                                  fontWeight:
                                                                      FontWeight
                                                                          .w600)),
                                                          SizedBox(height: 28),
                                                          Text(
                                                              'The reference reading surface, rendered from its original source.',
                                                              style: TextStyle(
                                                                  fontSize: 20,
                                                                  height: 1.5))
                                                        ])))),
                                        const SizedBox(height: 40)
                                      ])))))))));
  WidgetsBinding.instance.addPostFrameCallback((_) async {
    await Future<void>.delayed(const Duration(seconds: 2));
    await File('${Directory.systemTemp.path}/reference-ready.json')
        .writeAsString(jsonEncode({
      'route': route,
      'commit': '633680c84cec9a1b509f5092e8e2129562a6e047',
      'note': route == 'surface'
          ? 'Original surface component with comparison text'
          : 'Original screen and components, no Firebase initialization'
    }));
  });
}
